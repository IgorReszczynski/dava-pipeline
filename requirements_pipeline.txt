# --- Deep-learning framework ---------------------------------------------------
# NOTE: install a torch build that matches your NVIDIA driver. On the habra box
# the driver caps at CUDA 12.8, so if `import torch` complains the driver is
# "too old", reinstall with:  pip install torch torchaudio --index-url \
#   https://download.pytorch.org/whl/cu128
# CPU-only machine (Linux): use the smaller CPU build instead:
#   pip install torch==2.8.0 torchaudio==2.8.0 --index-url https://download.pytorch.org/whl/cpu
# (the default PyPI wheels on Windows and macOS are already CPU builds.)
torch==2.8.0
torchaudio==2.8.0

# --- ASR + forced alignment ----------------------------------------------------
whisperx==3.7.4
openai-whisper==20250625          # the --asr_backend openai engine
faster-whisper==1.2.1             # the --asr_backend faster-whisper engine (best on CPU)
ctranslate2==4.6.0                #   its inference runtime (int8 on CPU, fp16 on GPU)
transformers==4.57.1
accelerate==1.11.0
huggingface-hub==0.36.0

# --- Speaker diarization (needs a HF token for pyannote models) ----------------
pyannote.audio==3.4.0
pyannote.core==5.0.0
pyannote.database==5.1.3
pyannote.metrics==3.2.1
pyannote.pipeline==3.0.1

# torch>=2.6 pyannote 3.x checkpoints (WhisperX VAD + diarization) fail to load
# with "Weights only load failed ... omegaconf.listconfig.ListConfig".
lightning<2.6
pytorch-lightning<2.6
